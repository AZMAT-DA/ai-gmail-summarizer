Node.js Backend & API Setup Guide

Follow these steps to wrap your existing `index.js` CLI script into a full REST API server using Express.js or Next.js App Router.

1 Install Dependencies

Add Express & CORS to your existing directory:

npm install express cors
2 Create `server.js`

Expose a `/api/summaries` endpoint that streams the Gemini summaries.

3 Fetch from Frontend

Call `http://localhost:5000/api/summaries` directly in React/JS.

Production Ready `server.js` Code Snippet 
const express = require("express");
const cors = require("cors");
const fs = require("fs");
const path = require("path");
const { google } = require("googleapis");
const { GoogleGenAI } = require("@google/genai");
require("dotenv").config();

const app = express();
app.use(cors());
app.use(express.json());

const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });
const TOKEN_PATH = path.join(__dirname, "token.json");

app.get("/api/summaries", async (req, res) => {
  try {
    if (!fs.existsSync(TOKEN_PATH)) {
      return res.status(401).json({ error: "Authentication required. Run node index.js first." });
    }

    const credentials = JSON.parse(fs.readFileSync("credentials.json"));
    const { client_secret, client_id, redirect_uris } = credentials.installed || credentials.web;
    const oAuth2Client = new google.auth.OAuth2(client_id, client_secret, redirect_uris[0]);
    
    const token = fs.readFileSync(TOKEN_PATH);
    oAuth2Client.setCredentials(JSON.parse(token));

    const gmail = google.gmail({ version: "v1", auth: oAuth2Client });
    const listRes = await gmail.users.messages.list({ userId: "me", q: "is:unread", maxResults: 3 });

    const messages = listRes.data.messages || [];
    const results = [];

    for (const message of messages) {
      const msg = await gmail.users.messages.get({ userId: "me", id: message.id });
      const snippet = msg.data.snippet;
      const headers = msg.data.payload.headers;
      const subject = headers.find(h => h.name === "Subject")?.value || "No Subject";
      const sender = headers.find(h => h.name === "From")?.value || "Unknown Sender";

      const aiResponse = await ai.models.generateContent({
        model: "gemini-3.5-flash-lite",
        contents: `Summarize this email snippet concisely in 1-2 key points:\nSubject: ${subject}\nSnippet: ${snippet}`,
      });

      results.push({
        id: message.id,
        subject,
        sender,
        snippet,
        summary: aiResponse.text.trim()
      });
    }

    res.json({ success: true, count: results.length, data: results });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.listen(5000, () => console.log("🚀 Express Gmail AI Server running on http://localhost:5000"));
